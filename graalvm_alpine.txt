# Build stage with GraalVM
FROM ghcr.io/graalvm/native-image-community:21-ol9 AS builder

WORKDIR /build

# Install Maven
RUN microdnf install -y maven

# Copy dependency files first for better caching
COPY pom.xml .
COPY src ./src

# Build native image
RUN mvn -Pnative clean package -DskipTests

# Runtime stage - Alpine
FROM alpine:3.19

# Install only glibc (required for non-static binaries)
RUN apk add --no-cache \
    ca-certificates \
    tzdata \
    gcompat \
    libstdc++ && \
    addgroup -g 1001 appgroup && \
    adduser -D -u 1001 -G appgroup appuser

WORKDIR /app

# Copy the native binary
COPY --from=builder /build/target/*-runner /app/application

# Make binary executable
RUN chmod +x /app/application && \
    chown -R appuser:appgroup /app

USER appuser

EXPOSE 8080

# Health check (adjust endpoint as needed)
HEALTHCHECK --interval=30s --timeout=3s --start-period=5s --retries=3 \
    CMD wget --no-verbose --tries=1 --spider http://localhost:8080/q/health || exit 1

# Run the native application
ENTRYPOINT ["/app/application"]
