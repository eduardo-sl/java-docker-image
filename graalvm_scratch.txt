# Build stage with GraalVM
FROM ghcr.io/graalvm/native-image-community:21-muslib-ol9 AS builder

WORKDIR /build

# Install Maven
RUN microdnf install -y maven

# Copy dependency files first for better caching
COPY pom.xml .
COPY src ./src

# Build static native image (no external dependencies)
RUN mvn -Pnative clean package -DskipTests \
    -Dquarkus.native.additional-build-args=--static,--libc=musl \
    -Dquarkus.native.compression.level=10

# Runtime stage - Scratch (empty image)
FROM scratch

WORKDIR /app

# Copy the static native binary
COPY --from=builder /build/target/*-runner /app/application

# Copy CA certificates for HTTPS
COPY --from=builder /etc/pki/ca-trust/extracted/pem/tls-ca-bundle.pem /etc/ssl/certs/ca-certificates.crt

# Copy timezone data (optional)
COPY --from=builder /usr/share/zoneinfo /usr/share/zoneinfo

EXPOSE 8080

# Run the native application
ENTRYPOINT ["/app/application"]
