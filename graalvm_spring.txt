# Build stage with GraalVM for Spring Boot
FROM ghcr.io/graalvm/native-image-community:21-muslib-ol9 AS builder

WORKDIR /build

# Install Maven
RUN microdnf install -y maven

# Copy dependency files first for better caching
COPY pom.xml .
COPY src ./src

# Build Spring Boot native image with optimal settings
RUN mvn -Pnative clean package -DskipTests \
    -Dspring-boot.build-image.skip=true \
    native:compile \
    -Dspring.native.remove-unused-autoconfig=true \
    -Dspring.native.remove-yaml-support=false

# Runtime stage - Distroless static
FROM gcr.io/distroless/static-debian12:nonroot

WORKDIR /app

# Copy the native binary
COPY --from=builder /build/target/*-exec /app/application

# Spring Boot common ports
EXPOSE 8080
EXPOSE 8081

# Run the native application
ENTRYPOINT ["/app/application"]

# Optional: Add Spring Boot specific environment variables
# ENV SPRING_PROFILES_ACTIVE=prod
# ENV SERVER_PORT=8080
